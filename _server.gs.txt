function doGet(e) {
  try {
    const action = (e && e.parameter && e.parameter.action) || 'catalog';
    if (action === 'catalog') return json(getPublicCatalog());
    if (action === 'ping') return json({ ok: true });
    return json({ ok: false, error: 'Неизвестное действие' });
  } catch (err) {
    return json({ ok: false, error: String(err.message || err) });
  }
}

function doPost(e) {
  let body;
  try {
    body = JSON.parse(e.postData.contents);
  } catch (err) {
    return json({ ok: false, error: 'Некорректный запрос' });
  }
  try {
    const pub = PUBLIC_ACTIONS[body.action];
    if (pub) return json(pub(body));
    const adm = ADMIN_ACTIONS[body.action];
    if (adm) {
      requireAdmin(body.token);
      const result = withLock(() => adm(body));
      // Свежие данные — в том же ответе, чтобы админке не нужен был второй запрос.
      if (body.with_data && body.action !== 'admin_data') result.data = getAdminData();
      return json(result);
    }
    return json({ ok: false, error: 'Неизвестное действие' });
  } catch (err) {
    return json({ ok: false, error: String(err.message || err), auth: err.auth || false });
  }
}

const PUBLIC_ACTIONS = {
  order: body => withLock(() => createOrder(body)),
  promo: body => checkPromoPublic(body),
  events: body => logEvents(body),
  login: body => login(body)
};

const ADMIN_ACTIONS = {
  admin_data: () => getAdminData(),
  save_product: b => saveProduct(b.product),

function getPublicCatalog() {
  const cache = CacheService.getScriptCache();
  const hit = cache.get('catalog');
  if (hit) return JSON.parse(hit);

  const settings = readSettings();
  PRIVATE_SETTINGS.forEach(k => delete settings[k]);
  const sections = readTable('sections').rows
    .filter(s => s.name && bool(s.visible))
    .sort((a, b) => num(a.order) - num(b.order))
    .map(s => String(s.name).trim());
  const faq = readTable('faq').rows.filter(f => f.q).map(f => ({ q: String(f.q), a: String(f.a || '') }));
  const products = readTable('products').rows.filter(p => p.id && p.name).map(publicProduct);
  const sets = readSets().filter(s => s.visible);
  const reviews = readTable('reviews').rows.filter(r => r.text && bool(r.visible))
    .map(r => ({ name: str(r.name), text: str(r.text), photo: str(r.photo), date: dateStr(r.date) }));

  const data = { ok: true, store: settings, sections: sections, faq: faq, products: products, sets: sets, reviews: reviews, updated: nowStr() };
  try { cache.put('catalog', JSON.stringify(data), 600); } catch (e) { /* слишком большой — без кэша */ }
  return data;
}

function publicProduct(p) {
  const price = num(p.price);
  const sale = activeSale(p);
  return {
    id: String(p.id), brand: str(p.brand), name: str(p.name), gender: str(p.gender) || 'Унисекс',
    price: price, sale_percent: sale, sale_price: sale ? unitPrice(p) : null, sale_until: sale ? dateStr(p.sale_until) : '',
    in_stock: bool(p.in_stock), sections: splitList(p.sections), photo: str(p.photo),
    description: str(p.description), notes: str(p.notes), created: dateStr(p.created)
  };
}

function activeSale(p) {
  const pct = num(p.sale_percent);
  if (pct <= 0 || pct >= 100) return 0;
  const until = dateStr(p.sale_until);
  if (until && until < todayStr()) return 0;
  return pct;
}

function unitPrice(p) {
  const pct = activeSale(p);
  const price = num(p.price);
  return pct ? Math.round(price * (100 - pct) / 100) : price;
}

function createOrder(b) {
  rateLimit('orders_min', 30, 60);
  if (!b.consent) throw new Error('Нужно согласие на обработку персональных данных');
  const phone = normPhone(b.phone);
  if (!phone) throw new Error('Проверьте номер телефона');
  rateLimit('orders_' + phone, 6, 3600);
  const name = clean(b.name, 40), surname = clean(b.surname, 40);
  if (!name) throw new Error('Укажите имя');

  const settings = readSettings();
  const allowedMl = splitList(settings.ml_options).map(Number);
  const products = indexBy(readTable('products').rows, 'id');
  const rawItems = Array.isArray(b.items) ? b.items.slice(0, 30) : [];
  if (!rawItems.length) throw new Error('Корзина пуста');

  const sets = indexBy(readSets(), 'name');
  const giftAmounts = splitList(settings.gift_cards).map(Number);
  const items = rawItems.map(it => {
    const qty = Math.max(1, Math.min(10, Math.floor(Number(it.qty) || 1)));
    if (it.gift) {
      const amount = Number(it.gift);
      if (giftAmounts.indexOf(amount) < 0) throw new Error('Недоступный номинал сертификата');
      return { id: 'gift', gift: true, name: 'Подарочный сертификат ' + amount + ' ₸', ml: 0, qty: qty, price: amount };
    }
    const p = products[String(it.id)];
    if (!p) throw new Error('Аромат не найден — обновите страницу');
    if (!bool(p.in_stock)) throw new Error(fullName(p) + ' — нет в наличии');
    const ml = Number(it.ml);
    if (it.set) {
      const set = sets[String(it.set)];
      if (!set || !set.visible || set.items.indexOf(String(p.id)) < 0 || ml !== set.ml) throw new Error('Набор изменился — обновите страницу');
      return { id: String(p.id), name: fullName(p), ml: ml, qty: 1, set: set.name, price: Math.round(unitPrice(p) * (100 - set.discount) / 100) };
    }
    if (allowedMl.length && allowedMl.indexOf(ml) < 0) throw new Error('Недопустимый объём');
    return { id: String(p.id), name: fullName(p), ml: ml, qty: qty, price: unitPrice(p) };
  });
  // Скидка набора действует, только если в заказе весь набор целиком.
  Object.keys(sets).forEach(name => {
    const got = items.filter(it => it.set === name).map(it => it.id);
    if (got.length && sets[name].items.some(id => got.indexOf(id) < 0)) throw new Error('Набор «' + name + '» неполный — добавьте его заново');
  });

  const subtotal = items.reduce((s, it) => s + lineTotal(it), 0);
  let promoCode = '', discount = 0;
  if (b.promo) {
    const res = evalPromo(b.promo, phone, subtotal);
    if (!res.ok) throw new Error(res.error);
    promoCode = res.code; discount = res.discount;
  }

  // Номер заказа генерирует сайт, чтобы сразу открыть WhatsApp, не дожидаясь сервера.
  let number = String(b.number || '').toUpperCase();
  if (/^[A-Z0-9]{5}$/.test(number)) {
    const dup = readTable('orders').rows.find(o => str(o.number) === number);
    if (dup) {
      if (normPhone(dup.phone) === phone) return { ok: true, number: number, duplicate: true, total: num(dup.total) };
      number = nextOrderNumber();
    }
  } else {
    number = nextOrderNumber();
  }
  upsertClient(phone, name, surname, 'сайт');
  const order = {
    number: number, date: nowStr(), status: 'Новый', phone: phone, name: name, surname: surname,
    items_text: itemsText(items), subtotal: subtotal, promo: promoCode, discount: discount,
    total: subtotal - discount, comment: clean(b.comment, 300), source: 'сайт', updated: nowStr(),
    items: JSON.stringify(items)
  };
  appendRow('orders', order);
  return { ok: true, number: number, items: items, subtotal: subtotal, discount: discount, promo: promoCode, total: order.total };
}

function lineTotal(it) {
  return it.gift ? it.price * (it.qty || 1) : it.price * it.ml * (it.qty || 1);
}

function readSets() {
  return readTable('sets').rows.filter(s => s.name).map(s => ({
    name: str(s.name), items: splitList(s.items), ml: num(s.ml) || 3, discount: Math.min(90, Math.max(0, num(s.discount))),
    description: str(s.description), visible: bool(s.visible)
  }));
}

function checkPromoPublic(b) {
  rateLimit('promo_min', 60, 60);
  const phone = normPhone(b.phone);
  const res = evalPromo(b.code, phone, num(b.subtotal));
  if (!res.ok) return res;
  return { ok: true, code: res.code, type: res.type, value: res.value, discount: res.discount };
}

function evalPromo(code, phone, subtotal) {
  code = String(code || '').trim().toUpperCase();
  if (!code) return { ok: false, error: 'Введите промокод' };
  const p = readTable('promos').rows.find(r => String(r.code).trim().toUpperCase() === code);
  if (!p || !bool(p.active)) return { ok: false, error: 'Промокод не найден' };
  const until = dateStr(p.until);
  if (until && until < todayStr()) return { ok: false, error: 'Срок действия промокода истёк' };
  if (str(p.limit) !== '' && num(p.used) >= num(p.limit)) return { ok: false, error: 'Промокод уже использован' };
  const owner = normPhone(p.phone);
  if (owner && owner !== phone) return { ok: false, error: owner && !phone ? 'Укажите телефон, на который выдан промокод' : 'Этот промокод выдан другому клиенту' };
  const type = str(p.type) === '₸' ? '₸' : '%';
  const value = num(p.value);
  const discount = type === '%' ? Math.round(subtotal * Math.min(value, 100) / 100) : Math.min(value, subtotal);
  return { ok: true, code: code, type: type, value: value, discount: discount };
}

function logEvents(b) {
  rateLimit('events_min', 300, 60);
  const allowed = { view: 1, cart: 1, wa: 1, search: 1 };
  const list = (Array.isArray(b.events) ? b.events : []).slice(0, 25).filter(ev => allowed[ev.type]);
  if (!list.length) return { ok: true };
  const session = clean(b.session, 20);
  const now = nowStr();
  withLock(() => appendRows('events', list.map(ev => ({
    date: now, type: ev.type, product_id: clean(ev.id, 10), value: clean(ev.value, 60), session: session
  }))));
  return { ok: true };
}

function login(b) {
  const props = PropertiesService.getScriptProperties();
  const cache = CacheService.getScriptCache();
  const fails = Number(cache.get('login_fails') || 0);
  if (fails >= 10) throw new Error('Слишком много попыток. Подождите 15 минут.');
  const pass = String(b.password || '');
  const hash = props.getProperty('admin_hash');
  if (!hash) throw new Error('Пароль ещё не задан. Откройте таблицу → меню SMELL PARFUME → «Задать пароль админки».');
  if (hashPass(pass, props.getProperty('admin_salt')) !== hash) {
    cache.put('login_fails', String(fails + 1), 900);
    throw new Error('Неверный пароль');
  }
  return { ok: true, token: issueToken(), data: getAdminData() };
}

function passwordIsSet() {
  return !!PropertiesService.getScriptProperties().getProperty('admin_hash');
}

function setPassword(pass) {
  const props = PropertiesService.getScriptProperties();
  const salt = Utilities.getUuid();
  props.setProperty('admin_salt', salt);
  props.setProperty('admin_hash', hashPass(pass, salt));
  props.setProperty('admin_tokens', '{}');
}

function changePassword(b) {
  const props = PropertiesService.getScriptProperties();
  if (hashPass(String(b.old_password || ''), props.getProperty('admin_salt')) !== props.getProperty('admin_hash')) {
    throw new Error('Текущий пароль неверный');
  }
  if (String(b.new_password || '').length < 6) throw new Error('Новый пароль — не короче 6 символов');
  setPassword(String(b.new_password));
  return { ok: true, token: issueToken() };
}

function hashPass(pass, salt) {
  const bytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, salt + '|' + pass, Utilities.Charset.UTF_8);
  return Utilities.base64Encode(bytes);
}

function issueToken() {
  const props = PropertiesService.getScriptProperties();
  const tokens = pruneTokens(JSON.parse(props.getProperty('admin_tokens') || '{}'));
  const token = Utilities.getUuid() + Utilities.getUuid();
  tokens[token] = Date.now() + TOKEN_DAYS * 864e5;
  props.setProperty('admin_tokens', JSON.stringify(tokens));
  return token;
}

function pruneTokens(tokens) {
  const now = Date.now();
  Object.keys(tokens).forEach(t => { if (tokens[t] < now) delete tokens[t]; });
  return tokens;
}

function requireAdmin(token) {
  const tokens = JSON.parse(PropertiesService.getScriptProperties().getProperty('admin_tokens') || '{}');
  if (!token || !tokens[token] || tokens[token] < Date.now()) {
    const err = new Error('Войдите заново');
    err.auth = true;
    throw err;
  }
}

function logout(token) {
  const props = PropertiesService.getScriptProperties();
  const tokens = JSON.parse(props.getProperty('admin_tokens') || '{}');
  delete tokens[token];
  props.setProperty('admin_tokens', JSON.stringify(tokens));
  return { ok: true };
}

function getAdminData() {
  const settings = readSettings();
  const days = num(settings.no_answer_days) || 3;
  const limit = Utilities.formatDate(new Date(Date.now() - days * 864e5), TZ, 'yyyy-MM-dd HH:mm');

  const orders = readTable('orders');
  orders.rows.forEach(o => {
    if (o.status === 'Новый' && dateTimeStr(o.date) && dateTimeStr(o.date) < limit) {
      o.status = 'Без ответа';
      writeRow('orders', o);
    }
  });

  return {
    ok: true,
    password_set: passwordIsSet(),
    settings: settings,
    products: readTable('products').rows.map(p => Object.assign(publicProduct(p), {
      cost: num(p.cost), raw_price: num(p.price), sale_percent_raw: num(p.sale_percent), sale_until_raw: dateStr(p.sale_until)
    })),
    orders: orders.rows.map(o => ({
      number: str(o.number), date: dateTimeStr(o.date), status: str(o.status), phone: normPhone(o.phone) || str(o.phone),
      name: str(o.name), surname: str(o.surname), items_text: str(o.items_text), items: parseJson(o.items, []),
      subtotal: num(o.subtotal), promo: str(o.promo), discount: num(o.discount), total: num(o.total),
      comment: str(o.comment), source: str(o.source), updated: dateTimeStr(o.updated), gift_codes: str(o.gift_codes)
    })).reverse(),
    sets: readSets(),
    reviews: readTable('reviews').rows.filter(r => r.text || r.name).map(r => ({ name: str(r.name), text: str(r.text), photo: str(r.photo), date: dateStr(r.date), visible: bool(r.visible) })),
    sales: readTable('sales').rows.map(s => ({
      date: dateTimeStr(s.date), order: str(s.order), phone: normPhone(s.phone) || '', client: str(s.client),
      product_id: str(s.product_id), product: str(s.product), ml: num(s.ml), price: num(s.price), amount: num(s.amount),
      discount: num(s.discount), revenue: num(s.revenue), cost: num(s.cost), profit: num(s.profit)
    })),
    clients: readTable('clients').rows.filter(c => c.phone).map(c => ({
      phone: normPhone(c.phone) || str(c.phone), name: str(c.name), surname: str(c.surname),
      created: dateTimeStr(c.created), notes: str(c.notes), source: str(c.source)
    })),
    promos: readTable('promos').rows.filter(p => p.code).map(p => ({
      code: str(p.code), type: str(p.type) === '₸' ? '₸' : '%', value: num(p.value), until: dateStr(p.until),
      limit: str(p.limit) === '' ? '' : num(p.limit), used: num(p.used), phone: normPhone(p.phone) || '',
      active: bool(p.active), comment: str(p.comment), created: dateTimeStr(p.created)
    })),
    sections: readTable('sections').rows.filter(s => s.name).map(s => ({ name: str(s.name), order: num(s.order), visible: bool(s.visible) })),
    faq: readTable('faq').rows.filter(f => f.q).map(f => ({ q: str(f.q), a: str(f.a) })),
    events: aggregateEvents()
  };
}

function aggregateEvents() {
  const byProduct = {}, byDay = {};
  readTable('events').rows.forEach(ev => {
    const day = dateStr(ev.date);
    const type = str(ev.type);
    byDay[day] = byDay[day] || { view: 0, cart: 0, wa: 0, sessions: {} };
    if (byDay[day][type] !== undefined) byDay[day][type]++;
    if (ev.session) byDay[day].sessions[ev.session] = 1;
    if (ev.product_id) {
      const k = str(ev.product_id);
      byProduct[k] = byProduct[k] || { view: 0, cart: 0 };
      if (type === 'view' || type === 'cart') byProduct[k][type]++;
    }
  });
  Object.keys(byDay).forEach(d => { byDay[d].visitors = Object.keys(byDay[d].sessions).length; delete byDay[d].sessions; });
  return { byProduct: byProduct, byDay: byDay };
}

function saveProduct(p) {
  if (!p || !str(p.name)) throw new Error('Укажите название');
  const table = readTable('products');
  let row = p.id ? table.rows.find(r => str(r.id) === str(p.id)) : null;
  if (!row) {
    const max = table.rows.reduce((m, r) => Math.max(m, parseInt(String(r.id).replace(/\D/g, ''), 10) || 0), 0);
    row = { id: 'p' + String(max + 1).padStart(3, '0'), created: todayStr() };
  }
  Object.assign(row, {
    brand: clean(p.brand, 80), name: clean(p.name, 120), gender: ['Мужской', 'Женский', 'Унисекс'].indexOf(p.gender) >= 0 ? p.gender : 'Унисекс',
    price: num(p.price), cost: num(p.cost), in_stock: !!p.in_stock,
    sections: (Array.isArray(p.sections) ? p.sections : splitList(p.sections)).join(', '),
    sale_percent: num(p.sale_percent) || '', sale_until: str(p.sale_until),
    photo: str(p.photo), description: clean(p.description, 2000), notes: clean(p.notes, 500)
  });
  if (row._row) writeRow('products', row); else appendRow('products', row);
  bustCache();
  return { ok: true, id: row.id };
}

function deleteProduct(id) {
  const table = readTable('products');
  const row = table.rows.find(r => str(r.id) === str(id));
  if (!row) throw new Error('Аромат не найден');
  table.sheet.deleteRow(row._row);
  dirty('products');
  bustCache();
  return { ok: true };
}

function uploadPhoto(b) {
  if (!b.data) throw new Error('Нет файла');
  const mime = /^image\/(jpeg|png|webp)$/.test(b.mime) ? b.mime : 'image/jpeg';
  const blob = Utilities.newBlob(Utilities.base64Decode(b.data), mime, (b.id || 'photo') + '_' + Date.now());
  const file = photoFolder().createFile(blob);
  file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  const url = 'https://lh3.googleusercontent.com/d/' + file.getId() + '=w900';

  if (b.id) {
    const table = readTable('products');
    const row = table.rows.find(r => str(r.id) === str(b.id));
    if (row) {
      trashOwnPhoto(row.photo);
      row.photo = url;
      writeRow('products', row);
      bustCache();
    }
  }
  return { ok: true, url: url };
}

function photoFolder() {
  const props = PropertiesService.getScriptProperties();
  const id = props.getProperty('photo_folder');
  if (id) {
    try { return DriveApp.getFolderById(id); } catch (e) { /* папку удалили — создадим заново */ }
  }
  const folder = DriveApp.createFolder(PHOTO_FOLDER_NAME);
  props.setProperty('photo_folder', folder.getId());
  return folder;
}

function trashOwnPhoto(url) {
  const m = String(url || '').match(/googleusercontent\.com\/d\/([\w-]+)/);
  if (!m) return;
  try {
    const file = DriveApp.getFileById(m[1]);
    const parents = file.getParents();
    if (parents.hasNext() && parents.next().getId() === photoFolder().getId()) file.setTrashed(true);
  } catch (e) { /* не наш файл или уже удалён */ }
}

function setOrderStatus(number, status) {
  const allowed = ['Новый', 'Продано', 'Отменён', 'Без ответа'];
  if (allowed.indexOf(status) < 0) throw new Error('Неизвестный статус');
  const order = findOrder(number);
  const was = order.status;
  if (was === status) return { ok: true };
  if (status === 'Продано') { writeSales(order); issueGiftCodes(order); }
  if (was === 'Продано') { removeSales(order); revokeGiftCodes(order); }
  if (order.promo && (status === 'Продано') !== (was === 'Продано')) bumpPromo(order.promo, status === 'Продано' ? 1 : -1);
  order.status = status;
  order.updated = nowStr();
  writeRow('orders', order);
  return { ok: true };
}

// Проданный сертификат = промокод на сумму номинала, действует год, использовать можно один раз.
function issueGiftCodes(order) {
  const gifts = parseJson(order.items, []).filter(it => it.gift);
  if (!gifts.length) return;
  const abc = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
  const until = Utilities.formatDate(new Date(Date.now() + 365 * 864e5), TZ, 'yyyy-MM-dd');
  const codes = [];
  gifts.forEach(it => {
    for (let i = 0; i < (it.qty || 1); i++) {
      let code = 'GIFT-';
      for (let k = 0; k < 6; k++) code += abc[Math.floor(Math.random() * abc.length)];
      appendRow('promos', { code: code, type: '₸', value: it.price, until: until, limit: 1, used: 0, phone: '', active: true,
        comment: 'сертификат, заказ № ' + order.number, created: nowStr() });
      codes.push(code + ' (' + it.price + ' ₸)');
    }
  });
  order.gift_codes = codes.join(', ');
}

function revokeGiftCodes(order) {
  const codes = String(order.gift_codes || '').match(/GIFT-[A-Z0-9]{6}/g) || [];
  if (!codes.length) return;
  const table = readTable('promos');
  table.rows.filter(r => codes.indexOf(str(r.code)) >= 0).map(r => r._row)
    .sort((a, b) => b - a).forEach(r => table.sheet.deleteRow(r));
  dirty('promos');
  order.gift_codes = '';
}

function updateOrder(b) {
  const order = findOrder(b.number);
  if (order.status === 'Продано') throw new Error('Заказ уже проведён. Сначала верните его в «Новый».');
  const items = normalizeAdminItems(b.items);
  const subtotal = items.reduce((s, it) => s + lineTotal(it), 0);
  const discount = Math.min(Math.max(0, num(b.discount)), subtotal);
  Object.assign(order, {
    items: JSON.stringify(items), items_text: itemsText(items), subtotal: subtotal, discount: discount,
    total: subtotal - discount, comment: clean(b.comment, 300), updated: nowStr()
  });
  if (b.phone !== undefined) {
    const phone = normPhone(b.phone);
    if (phone) {
      order.phone = phone; order.name = clean(b.name, 40); order.surname = clean(b.surname, 40);
      upsertClient(phone, order.name, order.surname, 'вручную');
    }
  }
  writeRow('orders', order);
  return { ok: true };
}

function manualSale(b) {
  const items = normalizeAdminItems(b.items);
  const subtotal = items.reduce((s, it) => s + lineTotal(it), 0);
  const discount = Math.min(Math.max(0, num(b.discount)), subtotal);
  const phone = normPhone(b.phone);
  const name = clean(b.name, 40), surname = clean(b.surname, 40);
  if (phone) upsertClient(phone, name, surname, 'вручную');
  const date = b.date ? String(b.date).replace('T', ' ').slice(0, 16) : nowStr();
  const order = {
    number: nextOrderNumber(), date: date, status: 'Продано', phone: phone || '', name: name, surname: surname,
    items_text: itemsText(items), subtotal: subtotal, promo: '', discount: discount, total: subtotal - discount,
    comment: clean(b.comment, 300), source: 'вручную', updated: nowStr(), items: JSON.stringify(items)
  };
  appendRow('orders', order);
  writeSales(order);
  return { ok: true, number: order.number };
}

function normalizeAdminItems(list) {
  const products = indexBy(readTable('products').rows, 'id');
  const items = (Array.isArray(list) ? list : []).map(it => {
    if (it.gift) return num(it.price) > 0 ? { id: 'gift', gift: true, name: clean(it.name, 80) || 'Подарочный сертификат', ml: 0, qty: Math.max(1, Math.floor(num(it.qty) || 1)), price: num(it.price) } : null;
    const p = products[String(it.id)];
    const ml = num(it.ml);
    if (!p || ml <= 0) return null;
    return {
      id: String(p.id), name: fullName(p), ml: ml, qty: Math.max(1, Math.floor(num(it.qty) || 1)),
      price: str(it.price) === '' ? unitPrice(p) : num(it.price)
    };
  }).filter(Boolean);
  if (!items.length) throw new Error('Добавьте хотя бы один аромат');
  return items;
}

function writeSales(order) {
  const items = parseJson(order.items, []);
  const products = indexBy(readTable('products').rows, 'id');
  const subtotal = num(order.subtotal) || 1;
  const discount = num(order.discount);
  const client = [order.name, order.surname].filter(Boolean).join(' ');
  const date = dateTimeStr(order.date) || nowStr();
  let left = discount;
  const rows = items.map((it, i) => {
    const ml = it.ml * (it.qty || 1);
    const amount = lineTotal(it);
    const share = i === items.length - 1 ? left : Math.round(discount * amount / subtotal);
    left -= share;
    const p = products[it.id];
    const cost = Math.round((p ? num(p.cost) : 0) * ml * 100) / 100;
    const revenue = amount - share;
    return {
      date: date, order: str(order.number), phone: normPhone(order.phone) || '', client: client,
      product_id: it.id, product: it.name, ml: ml, price: it.price, amount: amount, discount: share,
      revenue: revenue, cost: cost, profit: Math.round((revenue - cost) * 100) / 100
    };
  });
  appendRows('sales', rows);
}

function removeSales(order) {
  const table = readTable('sales');
  table.rows.filter(r => str(r.order) === str(order.number)).map(r => r._row)
    .sort((a, b) => b - a).forEach(r => table.sheet.deleteRow(r));
  dirty('sales');
}

function findOrder(number) {
  const order = readTable('orders').rows.find(o => str(o.number) === str(number));
  if (!order) throw new Error('Заказ не найден');
  return order;
}

function saveClient(c, oldPhone) {
  const phone = normPhone(c && c.phone);
  if (!phone) throw new Error('Проверьте номер телефона');
  const table = readTable('clients');
  const key = normPhone(oldPhone) || phone;
  let row = table.rows.find(r => normPhone(r.phone) === key);
  if (key !== phone && table.rows.some(r => normPhone(r.phone) === phone)) throw new Error('Клиент с таким номером уже есть');
  if (!row) {
    if (table.rows.some(r => normPhone(r.phone) === phone)) throw new Error('Клиент с таким номером уже есть');
    row = { created: nowStr(), source: 'вручную' };
  }
  Object.assign(row, { phone: phone, name: clean(c.name, 40), surname: clean(c.surname, 40), notes: clean(c.notes, 1000) });
  if (row._row) writeRow('clients', row); else appendRow('clients', row);
  return { ok: true };
}

function deleteClient(phone) {
  const table = readTable('clients');
  const row = table.rows.find(r => normPhone(r.phone) === normPhone(phone));
  if (!row) throw new Error('Клиент не найден');
  table.sheet.deleteRow(row._row);
  dirty('clients');
  return { ok: true };
}

function upsertClient(phone, name, surname, source) {
  const table = readTable('clients');
  const row = table.rows.find(r => normPhone(r.phone) === phone);
  if (!row) {
    appendRow('clients', { phone: phone, name: name, surname: surname, created: nowStr(), notes: '', source: source });
  } else if ((!row.name && name) || (!row.surname && surname)) {
    row.name = row.name || name;
    row.surname = row.surname || surname;
    writeRow('clients', row);
  }
}

function savePromo(p, oldCode) {
  const code = String(p && p.code || '').trim().toUpperCase().replace(/\s+/g, '');
  if (!/^[A-ZА-ЯЁ0-9_-]{3,30}$/.test(code)) throw new Error('Код: 3–30 букв или цифр без пробелов');
  if (num(p.value) <= 0) throw new Error('Укажите размер скидки');
  const table = readTable('promos');
  const key = String(oldCode || code).toUpperCase();
  let row = table.rows.find(r => str(r.code).toUpperCase() === key);
  if (table.rows.some(r => str(r.code).toUpperCase() === code && r !== row)) throw new Error('Такой промокод уже есть');
  if (!row) row = { used: 0, created: nowStr() };
  Object.assign(row, {
    code: code, type: p.type === '₸' ? '₸' : '%', value: num(p.value), until: str(p.until),
    limit: str(p.limit) === '' ? '' : num(p.limit), phone: normPhone(p.phone) || '',
    active: p.active !== false, comment: clean(p.comment, 200)
  });
  if (row._row) writeRow('promos', row); else appendRow('promos', row);
  return { ok: true, code: code };
}

function deletePromo(code) {
  const table = readTable('promos');
  const row = table.rows.find(r => str(r.code).toUpperCase() === String(code).toUpperCase());
  if (!row) throw new Error('Промокод не найден');
  table.sheet.deleteRow(row._row);
  dirty('promos');
  return { ok: true };
}

function bumpPromo(code, delta) {
  const table = readTable('promos');
  const row = table.rows.find(r => str(r.code).toUpperCase() === String(code).toUpperCase());
  if (!row) return;
  row.used = Math.max(0, num(row.used) + delta);
  writeRow('promos', row);
}

function saveList(key, list) {
  if (!Array.isArray(list)) throw new Error('Нет данных');
  const table = readTable(key);
  const cols = Object.keys(SCHEMA[key].cols);
  const last = table.sheet.getLastRow();
  if (last > 1) table.sheet.getRange(2, 1, last - 1, table.headers.length).clearContent();
  dirty(key);
  const rows = list.map((item, i) => {
    const o = {};
    cols.forEach(c => { o[c] = item[c] === undefined ? '' : item[c]; });
    if (key === 'sections') { o.name = clean(o.name, 40); o.order = i + 1; o.visible = item.visible !== false; }
    return o;
  }).filter(o => str(o[cols[0]]));
  appendRows(key, rows);
  bustCache();
  return { ok: true };
}

function saveSettings(values) {
  const table = readTable('settings');
  Object.keys(values || {}).forEach(k => {
    const row = table.rows.find(r => str(r.key) === k);
    const v = String(values[k] === undefined ? '' : values[k]);
    if (row) { row.value = v; writeRow('settings', row); }
    else appendRow('settings', { key: k, value: v, desc: '' });
  });
  bustCache();
  return { ok: true };
}

function sheetOf(key) {
  const def = SCHEMA[key];
  const ss = SpreadsheetApp.getActive();
  let sh = ss.getSheetByName(def.sheet);
  if (!sh) {
    sh = ss.insertSheet(def.sheet);
    sh.getRange(1, 1, 1, Object.keys(def.cols).length).setValues([Object.values(def.cols)]).setFontWeight('bold');
    sh.setFrozenRows(1);
  }
  return sh;
}

// Кэш на время одного запроса: каждый лист читается из таблицы не больше одного раза.
const TABLES = {};
function dirty(key) { delete TABLES[key]; }

function readTable(key) {
  if (TABLES[key]) return TABLES[key];
  const def = SCHEMA[key];
  const sheet = sheetOf(key);
  const values = sheet.getDataRange().getValues();
  let headers = (values[0] || []).map(h => String(h).trim());
  // Недостающие колонки дописываем справа — так таблицу можно безопасно расширять.
  const missing = Object.values(def.cols).filter(h => headers.indexOf(h) < 0);
  if (missing.length) {
    sheet.getRange(1, headers.length + 1, 1, missing.length).setValues([missing]).setFontWeight('bold');
    headers = headers.concat(missing);
  }
  const idx = {};
  Object.keys(def.cols).forEach(k => { idx[k] = headers.indexOf(def.cols[k]); });
  const rows = [];
  for (let i = 1; i < values.length; i++) {
    const r = values[i];
    if (r.every(v => v === '' || v === null)) continue;
    const o = { _row: i + 1 };
    Object.keys(idx).forEach(k => { o[k] = idx[k] >= 0 && idx[k] < r.length ? r[idx[k]] : ''; });
    rows.push(o);
  }
  TABLES[key] = { sheet: sheet, headers: headers, idx: idx, rows: rows };
  return TABLES[key];
}

function toRow(key, obj, headers, existing) {
  const def = SCHEMA[key];
  const row = existing ? existing.slice() : headers.map(() => '');
  Object.keys(def.cols).forEach(k => {
    if (obj[k] === undefined) return;
    const i = headers.indexOf(def.cols[k]);
    if (i >= 0) row[i] = obj[k];
  });
  return row;
}

function appendRow(key, obj) { appendRows(key, [obj]); }

function appendRows(key, list) {
  if (!list.length) return;
  const sheet = sheetOf(key);
  const headers = readTable(key).headers;
  const start = sheet.getLastRow() + 1;
  const range = sheet.getRange(start, 1, list.length, headers.length);
  setTextFormats(key, sheet, headers, start, list.length);
  range.setValues(list.map(o => toRow(key, o, headers)));
  dirty(key);
}

// Строки в кэше — те же объекты, что меняет вызывающий код, поэтому кэш после записи остаётся верным.
function writeRow(key, obj) {
  const table = readTable(key);
  const sheet = table.sheet, headers = table.headers;
  const range = sheet.getRange(obj._row, 1, 1, headers.length);
  setTextFormats(key, sheet, headers, obj._row, 1);
  range.setValues([toRow(key, obj, headers, range.getValues()[0])]);
}

function setTextFormats(key, sheet, headers, start, count) {
  (TEXT_COLS[key] || []).forEach(k => {
    const i = headers.indexOf(SCHEMA[key].cols[k]);
    if (i >= 0) sheet.getRange(start, i + 1, count, 1).setNumberFormat('@');
  });
}

function readSettings() {
  const out = {};
  readTable('settings').rows.forEach(r => { if (r.key) out[str(r.key)] = str(r.value); });
  return out;
}

function withLock(fn) {
  const lock = LockService.getScriptLock();
  lock.waitLock(25000);
  try { return fn(); } finally { lock.releaseLock(); }
}

function rateLimit(key, max, seconds) {
  const cache = CacheService.getScriptCache();
  const n = Number(cache.get('rl_' + key) || 0);
  if (n >= max) throw new Error('Слишком много запросов, попробуйте чуть позже');
  cache.put('rl_' + key, String(n + 1), seconds);
}

function bustCache() { CacheService.getScriptCache().remove('catalog'); }

function nextOrderNumber() {
  const props = PropertiesService.getScriptProperties();
  let n = Number(props.getProperty('order_seq') || 1000);
  const existing = readTable('orders').rows.reduce((m, o) => Math.max(m, Number(o.number) || 0), 0);
  n = Math.max(n, existing) + 1;
  props.setProperty('order_seq', String(n));
  return String(n);
}

function itemsText(items) {
  return items.map(it => it.gift
    ? it.name + (it.qty > 1 ? ' × ' + it.qty : '')
    : it.name + ' — ' + it.ml + ' мл' + (it.qty > 1 ? ' × ' + it.qty : '') + ' × ' + it.price + ' ₸/мл' + (it.set ? ' (набор «' + it.set + '»)' : '')).join('; ');
}

function fullName(p) { return [str(p.brand), str(p.name)].filter(Boolean).join(' '); }

function normPhone(v) {
  let d = String(v || '').replace(/\D/g, '');
  if (d.length === 11 && d[0] === '8') d = '7' + d.slice(1);
  if (d.length === 10) d = '7' + d;
  return /^7\d{10}$/.test(d) ? d : '';
}

function num(v) {
  if (typeof v === 'number') return v;
  const n = parseFloat(String(v || '').replace(/\s/g, '').replace(',', '.'));
  return isNaN(n) ? 0 : n;
}

function bool(v) { return v === true || /^(true|да|1|yes|истина)$/i.test(String(v).trim()); }
function str(v) { return v === null || v === undefined ? '' : (v instanceof Date ? dateStr(v) : String(v).trim()); }
function clean(v, max) { return String(v || '').replace(/[<>]/g, '').trim().slice(0, max || 200); }
function splitList(v) { return String(v || '').split(',').map(s => s.trim()).filter(Boolean); }
function indexBy(rows, key) { const o = {}; rows.forEach(r => { o[str(r[key])] = r; }); return o; }
function parseJson(v, def) { try { return JSON.parse(v); } catch (e) { return def; } }

function nowStr() { return Utilities.formatDate(new Date(), TZ, 'yyyy-MM-dd HH:mm'); }
function todayStr() { return Utilities.formatDate(new Date(), TZ, 'yyyy-MM-dd'); }

function dateStr(v) {
  if (!v) return '';
  if (v instanceof Date) return Utilities.formatDate(v, TZ, 'yyyy-MM-dd');
  const s = String(v).trim();
  const m = s.match(/^(\d{4})-(\d{2})-(\d{2})/) || s.match(/^(\d{2})\.(\d{2})\.(\d{4})/);
  if (!m) return '';
  return m[1].length === 4 ? m[1] + '-' + m[2] + '-' + m[3] : m[3] + '-' + m[2] + '-' + m[1];
}

function dateTimeStr(v) {
  if (!v) return '';
  if (v instanceof Date) return Utilities.formatDate(v, TZ, 'yyyy-MM-dd HH:mm');
  const s = String(v).trim();
  const d = dateStr(s);
  const t = s.match(/(\d{1,2}):(\d{2})/);
  return d ? d + ' ' + (t ? t[1].padStart(2, '0') + ':' + t[2] : '00:00') : '';
}
